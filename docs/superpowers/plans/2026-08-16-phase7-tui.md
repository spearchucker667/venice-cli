# Phase 7 — Full Interactive Terminal UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the simple console `AgentRenderer` with a keyboard-driven, full-screen Ink/React TUI for `venice agent` that renders a transcript, persistent composer, tool/approval events, streaming assistant text, and a status bar while keeping all business logic in the existing runtime.

**Architecture:** The TUI is a thin React layer over the existing `AgentRuntime` and `EventBus`. A top-level `App` component subscribes to runtime events, maintains transcript and status state, and renders `Transcript`, `Composer`, `Approval`, `ToolCall`, and `StatusBar` components. User input from `Composer` feeds back into the runtime through public runtime methods. The approver callback on `PermissionManager` is replaced with one that renders an `Approval` prompt and resolves when the user chooses. A renderer facade selects the Ink TUI in interactive TTY mode and the existing plain renderer otherwise.

**Tech Stack:** Node 18+, TypeScript strict mode, Ink, React, ink-text-input, ink-select-input, ink-testing-library, existing agent runtime.

## Global Constraints

- No weakening of TypeScript strict mode.
- Keep runtime/business logic out of UI components.
- Existing non-interactive behavior must remain unchanged.
- Add only the minimal set of UI dependencies required by Ink.
- Every new file must have tests.
- Do not auto-commit or auto-push; follow the repository’s no-commit policy.

```ts
export function parseSlashCommand(input: string): { command: string; args: string } | undefined {
  const trimmed = input.trim();
  if (!trimmed.startsWith('/')) return undefined;
  const [command, ...rest] = trimmed.slice(1).split(' ');
  return { command: command.toLowerCase(), args: rest.join(' ').trim() };
}
```

```ts
import * as path from 'node:path';

export interface MentionResolution {
  text: string;
  mentions: string[];
}

export function resolveMentions(input: string, workspaceRoot: string): MentionResolution {
  const mentions: string[] = [];
  const mentionPattern = /@(\S+)/g;
  const text = input.replace(mentionPattern, (_match, mention: string) => {
    const normalized = path.normalize(mention).replace(/^\.\.\/+/g, '').replace(/^\//, '');
    if (normalized && !mentions.includes(normalized)) {
      mentions.push(normalized);
    }
    return mention;
  });
  return { text, mentions };
}
```

```tsx
import React, { useState } from 'react';
import { Box, Text } from 'ink';
import TextInput from 'ink-text-input';

export interface ComposerProps {
  onSubmit: (text: string, mentions: string[]) => void;
  disabled?: boolean;
}

export function Composer({ onSubmit, disabled }: ComposerProps): JSX.Element {
  const [value, setValue] = useState('');

  const handleSubmit = () => {
    if (!value.trim() || disabled) return;
    onSubmit(value.trim(), []);
    setValue('');
  };

  return (
    <Box>
      <Text>{'> '}</Text>
      <TextInput value={value} onChange={setValue} onSubmit={handleSubmit} />
    </Box>
  );
}
```

```tsx
    { label: 'Yes', value: { approved: true, scope: 'once' as const } },

    { label: 'No', value: { approved: false } },

      <Text bold>Approve {toolName}?</Text>
      <Text dimColor>Risk: {risk}</Text>
      <Text>{JSON.stringify(input)}</Text>
      <SelectInput items={items} onSelect={(item) => onDecision(item.value)} />
```

```tsx
import { Box, Text } from 'ink';
import type { TuiMessage } from './types.js';
import { ToolCallEvent } from './tool-call.js';

export interface TranscriptProps {
  messages: TuiMessage[];
}

export function Transcript({ messages }: TranscriptProps): JSX.Element {
  return (
    <Box flexDirection="column" flexGrow={1}>
      {messages.map((message) => {
        switch (message.role) {
          case 'user':
            return (
              <Box key={message.id}>
                <Text bold>{'> '}</Text>
                <Text>{message.content}</Text>
              </Box>
            );
          case 'assistant':
            return (
              <Box key={message.id} flexDirection="column">
                <Text>{message.content}</Text>
              </Box>
            );
          case 'tool':
            return (
              <ToolCallEvent
                key={message.id}
                toolName={String(message.metadata?.toolName || 'tool')}
                input={message.metadata?.input}
                ok={message.metadata?.ok as boolean | undefined}
                error={message.metadata?.error as string | undefined}
              />
            );
          case 'event':
            return (
              <Box key={message.id}>
                <Text dimColor>{message.content}</Text>
              </Box>
            );
          default:
            return null;
        }
      })}
    </Box>
  );
}
```

```tsx
import React, { useEffect, useState } from 'react';
import { Box, useApp, useInput } from 'ink';
import type { AgentRuntime } from '../agent/runtime.js';
import type { AgentEvent, AgentStatus } from '../agent/types.js';
import { Composer } from './composer.js';
import { Transcript } from './transcript.js';
import { StatusBar } from './status.js';
import { ApprovalPrompt, type ApprovalDecision } from './approval.js';
import type { TuiMessage, TuiState } from './types.js';

export interface AppProps {
  runtime: AgentRuntime;
  onExit: () => void;
}

export function App({ runtime, onExit }: AppProps): JSX.Element {
  const { exit } = useApp();
  const [messages, setMessages] = useState<TuiMessage[]>([]);
  const [state, setState] = useState<TuiState>({
    messages: [],
    status: 'idle',
    model: runtime.getState().model,
    workspaceRoot: runtime.getState().workspaceRoot,
    approvalMode: 'suggest',
    contextTokens: 0,
    maxTokens: 128000,
  });
  const [pendingApproval, setPendingApproval] = useState<{
    resolve: (decision: ApprovalDecision) => void;
    toolName: string;
    input: unknown;
    risk: string;
  } | null>(null);
  const [composerDisabled, setComposerDisabled] = useState(false);

  useInput((_input, key) => {
    if (key.ctrl && key.name === 'c') {
      exit();
      onExit();
    }
  });

  useEffect(() => {
    runtime.getState();
    // TODO: wire event subscription and runtime run in follow-up task
  }, [runtime]);

  const handleSubmit = (text: string, _mentions: string[]) => {
    setMessages((prev) => [...prev, { id: String(prev.length + 1), role: 'user', content: text }]);
  };

  return (
    <Box flexDirection="column" height="100%">
      <Transcript messages={messages} />
      {pendingApproval && (
        <ApprovalPrompt
          toolName={pendingApproval.toolName}
          input={pendingApproval.input}
          risk={pendingApproval.risk}
          onDecision={(decision) => {
            pendingApproval.resolve(decision);
            setPendingApproval(null);
          }}
        />
      )}
      <Composer onSubmit={handleSubmit} disabled={composerDisabled || pendingApproval !== null} />
      <StatusBar state={state} />
    </Box>
  );
}
```

```ts
export function mapEventToMessage(event: AgentEvent): TuiMessage | undefined {
  switch (event.type) {
    case 'session_started':
      return { id: event.eventId, role: 'event', content: `▶ ${event.objective}` };
    case 'model_request':
      return { id: event.eventId, role: 'event', content: 'thinking…' };

    case 'tool_completed':
      return {

        role: 'tool',
        content: event.result.ok ? 'done' : 'failed',
        metadata: { toolName: event.toolName, input: event.input, ok: event.result.ok, error: event.result.error?.message },

    case 'approval_requested':
      return { id: event.eventId, role: 'event', content: `? approval required: ${event.toolName}` };
    case 'file_changed':
      return { id: event.eventId, role: 'event', content: `~ ${event.path}` };
    case 'context_compacted':
      return { id: event.eventId, role: 'event', content: '… context compacted' };
    case 'session_completed':
```
